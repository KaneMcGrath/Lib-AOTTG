cmake_minimum_required(VERSION 3.25)
project(lib-aottg-cpp VERSION 0.1.0 LANGUAGES CXX)
option(LIBAOTTG_BUILD_TESTS "Build native parity, reference, collision and benchmark tools" ON)
option(LIBAOTTG_ENABLE_IPO "Enable compiler/linker optimization across native translation units" OFF)
if(LIBAOTTG_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(NOT ipo_supported)
    message(FATAL_ERROR "Native IPO unavailable: ${ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
add_library(lib-aottg-cpp STATIC src/Core.cpp src/Hud.cpp src/HudPackage.cpp)
target_include_directories(lib-aottg-cpp PUBLIC include)
target_compile_features(lib-aottg-cpp PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(lib-aottg-cpp PUBLIC /EHsc /fp:precise PRIVATE /W4)
else()
  target_compile_options(lib-aottg-cpp PRIVATE -Wall -Wextra -ffp-contract=off)
endif()
if(LIBAOTTG_BUILD_TESTS)
add_executable(LibAottgHeroHudChecks tests/HeroHud.cpp)
target_link_libraries(LibAottgHeroHudChecks PRIVATE lib-aottg-cpp)
add_executable(LibAottgParity tests/Parity.cpp)
target_link_libraries(LibAottgParity PRIVATE lib-aottg-cpp)
add_executable(LibAottgReference tests/Reference.cpp)
target_link_libraries(LibAottgReference PRIVATE lib-aottg-cpp)
add_executable(LibAottgChecks tests/Checks.cpp)
target_link_libraries(LibAottgChecks PRIVATE lib-aottg-cpp)
add_executable(LibAottgBenchmark tests/Benchmark.cpp)
target_link_libraries(LibAottgBenchmark PRIVATE lib-aottg-cpp)
endif()

option(LIBAOTTG_BUILD_EXAMPLES "Build the minimal host example" OFF)
if(LIBAOTTG_BUILD_EXAMPLES)
  add_executable(LibAottgConsoleHost examples/ConsoleHost.cpp)
  target_link_libraries(LibAottgConsoleHost PRIVATE lib-aottg-cpp)
endif()
